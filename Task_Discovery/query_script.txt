// 1) Generic tasks
MATCH (gen:Task)
WHERE gen.category IN ['KPI', 'Assessment', 'Service', 'VirtualSensor']
  AND gen.taskId IS NOT NULL

// 2) Find instances with the same taskId
OPTIONAL MATCH (inst:Task)
WHERE inst.category IN [
        'KPI-Instance',
        'Assessment-Instance',
        'Service-Instance',
        'VirtualSensor-Instance'
      ]
  AND inst.taskId IS NOT NULL
  AND trim(inst.taskId) = trim(gen.taskId)

// Collect instances and compute hasInstance (without filtering them out)
WITH gen, collect(DISTINCT inst) AS instancesRaw
// remove NULLs from instances
WITH
  gen,
  [i IN instancesRaw WHERE i IS NOT NULL] AS instances
WITH
  gen,
  instances,
  size(instances) > 0 AS hasInstance

// 3) Requirements (direct + from required tasks)
OPTIONAL MATCH (gen)-[:requires]->(tgtTask:Task)
OPTIONAL MATCH (gen)-[:requires]->(req1:Requirement)
OPTIONAL MATCH (tgtTask)-[:requires]->(req2:Requirement)

WITH
  gen,
  instances,
  hasInstance,
  collect(DISTINCT trim(tgtTask.taskId)) AS requiresTaskIds,
  collect(
    DISTINCT {
      key: req1.key,
      fromTaskId: trim(gen.taskId)
    }
  ) AS directReqInfo,
  collect(
    DISTINCT {
      key: req2.key,
      fromTaskId: trim(tgtTask.taskId)
    }
  ) AS nestedReqInfo

// 4) Topology via reqAs* chains from instances
//    For tasks with no instances, we UNWIND [NULL] so we still keep a row.
WITH
  gen,
  instances,
  hasInstance,
  requiresTaskIds,
  directReqInfo,
  nestedReqInfo,
  CASE
    WHEN size(instances) > 0 THEN instances
    ELSE [NULL]
  END AS instList

UNWIND instList AS inst

// From each instance, walk along relationships whose type CONTAINS 'reqAs'
// until we hit a Building / Space / Zone
OPTIONAL MATCH p = (inst)-[rels*1..6]-(anchor)
WHERE inst IS NOT NULL
  AND all(rel IN rels WHERE type(rel) CONTAINS 'reqAs')
  AND (anchor:Building OR anchor:Space OR anchor:Zone)

WITH
  gen,
  hasInstance,
  requiresTaskIds,
  directReqInfo,
  nestedReqInfo,
  collect(DISTINCT anchor) AS anchorsRaw

// Keep row even if there are no anchors (use [NULL])
WITH
  gen,
  hasInstance,
  requiresTaskIds,
  directReqInfo,
  nestedReqInfo,
  CASE
    WHEN size(anchorsRaw) > 0 THEN anchorsRaw
    ELSE [NULL]
  END AS anchors

UNWIND anchors AS a

// From anchors, derive buildings/spaces/zones:
// - Buildings directly
// - Spaces directly, and via zones
// - Zones directly
OPTIONAL MATCH (a)-[:isSpaceOf|hasSpace]->(bFromSpace:Building)
OPTIONAL MATCH (a)-[:hasZone|isZoneOf]->(sFromZone:Space)
OPTIONAL MATCH (sFromZone)-[:isSpaceOf|hasSpace]->(bFromZSpace:Building)

WITH
  gen,
  hasInstance,
  requiresTaskIds,
  directReqInfo,
  nestedReqInfo,
  collect(DISTINCT CASE WHEN a:Building THEN a.id END) AS directBIds,
  collect(DISTINCT CASE WHEN a:Space    THEN a.id END) AS directSIds,
  collect(DISTINCT CASE WHEN a:Zone     THEN a.id END) AS directZIds,
  collect(DISTINCT bFromSpace.id)   AS bFromSpaceIds,
  collect(DISTINCT sFromZone.id)    AS sFromZoneIds,
  collect(DISTINCT bFromZSpace.id)  AS bFromZSpaceIds

// 5) Final aggregation per generic task
RETURN
  trim(gen.taskId)               AS genericTaskId,
  gen.category                   AS genericCategory,
  coalesce(gen.label, gen.name)  AS genericName,
  hasInstance,  // now correctly true or false
  [id IN requiresTaskIds WHERE id IS NOT NULL] AS requiresTaskIds,
  [info IN (directReqInfo + nestedReqInfo)
   WHERE info.key IS NOT NULL] AS requirementInfo,
  // Buildings: direct from anchors, or via space / zone→space
  [id IN (directBIds + bFromSpaceIds + bFromZSpaceIds)
   WHERE id IS NOT NULL] AS buildingIds,
  // Spaces: direct anchors, plus spaces reached from zones
  [id IN (directSIds + sFromZoneIds)
   WHERE id IS NOT NULL] AS spaceIds,
  // Zones: only direct anchors that are zones
  [id IN directZIds
   WHERE id IS NOT NULL] AS zoneIds
ORDER BY genericTaskId;
